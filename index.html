// ══════════════════════════════════════════════════════════ // auto-poster — Supabase Edge Function // Invocada a cada minuto pelo pg_cron. // Verifica quais NPCs têm post agendado para agora, // gera o conteúdo via Gemini e publica no banco. // ══════════════════════════════════════════════════════════ import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const supabase = createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")! ); // ── Mapa de tons ────────────────────────────────────────── const TONS: Record = { natural: "natural, condizente com a personalidade e bio do personagem", animado: "animado, empolgado, cheio de energia", poetico: "poético, contemplativo, com metáforas", misterioso: "misterioso, sombrio, enigmático", formal: "formal, sério, distinto", comico: "cômico, descontraído, com humor", }; // ── Hora atual em BRT (UTC-3) no formato "HH:MM" ───────── function horaAtualBRT(): string { const now = new Date(); const brt = new Date(now.getTime() - 3 * 60 * 60 * 1000); const h = brt.getUTCHours().toString().padStart(2, "0"); const m = brt.getUTCMinutes().toString().padStart(2, "0"); return `${h}:${m}`; } // ── Verifica se o NPC já postou nesta janela de 1 minuto ── function jaPostouNaJanela(ultimoPost: string | null, agora: Date): boolean { if (!ultimoPost) return false; const ultimo = new Date(ultimoPost); // Janela de 2 minutos para evitar post duplo em caso de retry return (agora.getTime() - ultimo.getTime()) < 2 * 60 * 1000; } // ── Gera texto via Gemini ───────────────────────────────── async function gerarTexto( geminiKey: string, tipo: string, npc: { username: string; name: string; bio: string; rpg: string }, tom: string, contexto: string | null ): Promise<{ texto?: string; legenda?: string; local?: string }> { const tomDesc = TONS[tom] || "natural"; const ctx = contexto || "Situação cotidiana do personagem no universo Pokémon"; let prompt: string; if (tipo === "pokegram") { prompt = `Você é um assistente criativo para um RPG de Pokémon chamado Pokégram. Gere UMA legenda para uma postagem de foto do personagem descrito. A legenda deve parecer uma postagem real de uma rede social estilo Instagram no universo Pokémon. Use português brasileiro informal. Inclua hashtags relevantes ao universo Pokémon no final. Responda APENAS com um JSON válido, sem texto extra, sem markdown. Personagem: @${npc.username}${npc.name ? " (" + npc.name + ")" : ""} Bio: ${npc.bio || "Sem bio definida"} RPG: ${npc.rpg || "Não definido"} Contexto: ${ctx} Tom: ${tomDesc} Responda exatamente neste formato: {"legenda":"texto com #hashtags","local":"Nome do Local, Região Pokémon"}`; } else { prompt = `Você é um assistente criativo para um RPG de Pokémon chamado Pokégram. Gere UM Pokeet (post curto estilo Twitter) do personagem descrito. O texto deve parecer um post real de uma rede social no universo Pokémon. Use português brasileiro informal. Máximo 280 caracteres. Responda APENAS com um JSON válido, sem texto extra, sem markdown. Personagem: @${npc.username}${npc.name ? " (" + npc.name + ")" : ""} Bio: ${npc.bio || "Sem bio definida"} RPG: ${npc.rpg || "Não definido"} Contexto: ${ctx} Tom: ${tomDesc} Responda exatamente neste formato: {"texto":"texto do pokeet (máx 280 chars)"}`; } const resp = await fetch( `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${geminiKey}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }], generationConfig: { temperature: 0.9, maxOutputTokens: 512, responseMimeType: "application/json", }, }), } ); if (!resp.ok) { const err = await resp.json().catch(() => ({})); throw new Error(`Gemini error: ${err?.error?.message || resp.status}`); } const data = await resp.json(); let raw: string = data?.candidates?.[0]?.content?.parts?.[0]?.text || ""; // Limpa markdown fences e extrai JSON raw = raw.replace(/```json\s*/gi, "").replace(/```/g, "").trim(); const s = raw.indexOf("{"); const e = raw.lastIndexOf("}"); if (s !== -1 && e !== -1) raw = raw.slice(s, e + 1); try { return JSON.parse(raw); } catch { // Tenta corrigir quebras de linha dentro de strings const fixed = raw.replace(/("(?:[^"\\]|\\.)*")/g, (m) => m.replace(/\n/g, " ").replace(/\r/g, "") ); return JSON.parse(fixed); } } // ── Publica no Pokégram ─────────────────────────────────── async function publicarPost( npcId: string, legenda: string, local: string | null ) { const { error } = await supabase.from("posts").insert({ user_id: npcId, caption: legenda, location: local || null, image_url: null, emoji: "📸", bg: "#f5f5f5", }); if (error) throw new Error(`Erro ao publicar post: ${error.message}`); } // ── Publica no Pokeet ───────────────────────────────────── async function publicarPokeet(npcId: string, texto: string) { const { error } = await supabase.from("pokeets").insert({ user_id: npcId, text: texto.slice(0, 280), }); if (error) throw new Error(`Erro ao publicar pokeet: ${error.message}`); } // ── Atualiza ultimo_post ────────────────────────────────── async function marcarUltimoPost(npcId: string) { await supabase .from("npc_autoposts") .update({ ultimo_post: new Date().toISOString() }) .eq("npc_id", npcId); } // ── Handler principal ───────────────────────────────────── Deno.serve(async (_req) => { const agora = new Date(); const horaHoje = horaAtualBRT(); const log: string[] = [`[auto-poster] ${agora.toISOString()} — hora BRT: ${horaHoje}`]; try { // Busca chave Gemini const { data: keyRow } = await supabase .from("settings") .select("value") .eq("key", "anthropic_api_key") .maybeSingle(); const geminiKey = keyRow?.value || ""; if (!geminiKey) { return new Response( JSON.stringify({ ok: false, error: "Chave Gemini não configurada em settings." }), { status: 200, headers: { "Content-Type": "application/json" } } ); } // Busca configs ativas const { data: configs, error: cfgErr } = await supabase .from("npc_autoposts") .select(` npc_id, tipo, tom, contexto_fixo, horarios, ultimo_post, users!npc_autoposts_npc_id_fkey (id, username, name, bio, rpg) `) .eq("ativo", true); if (cfgErr) throw new Error(`Erro ao buscar configs: ${cfgErr.message}`); if (!configs?.length) { return new Response( JSON.stringify({ ok: true, log: ["Nenhuma automação ativa."] }), { status: 200, headers: { "Content-Type": "application/json" } } ); } const resultados: { npc: string; acao: string }[] = []; for (const cfg of configs) { const npc = cfg.users as { id: string; username: string; name: string; bio: string; rpg: string; }; if (!npc) continue; // Verifica se a hora atual bate com algum horário agendado (tolerância ±1 min) const horarios: string[] = cfg.horarios || []; const devePostar = horarios.some((h: string) => h.trim() === horaHoje); if (!devePostar) continue; // Evita post duplo na mesma janela if (jaPostouNaJanela(cfg.ultimo_post, agora)) { log.push(`@${npc.username}: já postou nesta janela, pulando.`); continue; } // Decide tipo quando for "ambos" let tipo = cfg.tipo; if (tipo === "ambos") tipo = Math.random() < 0.5 ? "pokegram" : "pokeet"; try { const gerado = await gerarTexto( geminiKey, tipo, npc, cfg.tom, cfg.contexto_fixo ); if (tipo === "pokegram") { const legenda = gerado.legenda || ""; if (!legenda) throw new Error("Gemini não retornou legenda."); await publicarPost(npc.id, legenda, gerado.local || null); } else { const texto = gerado.texto || ""; if (!texto) throw new Error("Gemini não retornou texto."); await publicarPokeet(npc.id, texto); } await marcarUltimoPost(cfg.npc_id); const acao = `publicou ${tipo === "pokegram" ? "post" : "pokeet"} com sucesso`; log.push(`@${npc.username}: ${acao}`); resultados.push({ npc: npc.username, acao }); } catch (err) { const msg = err instanceof Error ? err.message : String(err); log.push(`@${npc.username}: ERRO — ${msg}`); resultados.push({ npc: npc.username, acao: `ERRO: ${msg}` }); } } return new Response( JSON.stringify({ ok: true, hora: horaHoje, log, resultados }), { status: 200, headers: { "Content-Type": "application/json" } } ); } catch (err) { const msg = err instanceof Error ? err.message : String(err); log.push(`ERRO GERAL: ${msg}`); return new Response( JSON.stringify({ ok: false, log }), { status: 500, headers: { "Content-Type": "application/json" } } ); } });
